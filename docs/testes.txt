```go
// internal/domain/pgrs/service/pgrs_service_test.go
package service

import (
    "context"
    "testing"
    "github.com/stretchr/testify/assert"
    "github.com/stretchr/testify/mock"
)

// Mock do Repository
type MockPGRSRepository struct {
    mock.Mock
}

func (m *MockPGRSRepository) Create(ctx context.Context, pgrs *models.PGRS) error {
    args := m.Called(ctx, pgrs)
    return args.Error(0)
}

// Teste
func TestPGRSService_CriarRascunho(t *testing.T) {
    // Arrange
    mockRepo := new(MockPGRSRepository)
    mockRepo.On("Create", mock.Anything, mock.Anything).Return(nil)
    
    service := NewPGRSService(mockRepo, nil, nil)
    
    // Act
    pgrs, err := service.CriarRascunho(context.Background(), &dto.CriarPGRSRequest{
        UnidadeID: 1,
    })
    
    // Assert
    assert.NoError(t, err)
    assert.NotNil(t, pgrs)
    assert.Equal(t, "RASCUNHO", pgrs.Status)
    mockRepo.AssertExpectations(t)
}
```

### Testando Repository com DB

```go
// internal/domain/pgrs/repository/pgrs_repo_integration_test.go
func TestPGRSRepository_Integration(t *testing.T) {
    if testing.Short() {
        t.Skip("Pulando teste de integração")
    }
    
    // Setup DB de teste
    db := setupTestDB(t)
    defer teardownTestDB(db)
    
    repo := NewPGRSRepository(db)
    
    // Teste
    pgrs := &models.PGRS{
        Numero: "TEST-001",
        Status: "RASCUNHO",
    }
    
    err := repo.Create(context.Background(), pgrs)
    assert.NoError(t, err)
    assert.NotZero(t, pgrs.ID)
    
    // Buscar
    found, err := repo.FindByID(context.Background(), pgrs.ID)
    assert.NoError(t, err)
    assert.Equal(t, "TEST-001", found.Numero)
}

func setupTestDB(t *testing.T) *gorm.DB {
    db, err := gorm.Open(sqlite.Open(":memory:"), &gorm.Config{})
    require.NoError(t, err)
    
    db.AutoMigrate(&models.PGRS{})
    return db
}
```

### CI/CD

```yaml
# .github/workflows/tests.yml
name: Tests

on: [push, pull_request]

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v2
      - uses: actions/setup-go@v2
      
      - name: Run tests
        run: go test ./... -cover
      
      - name: Check coverage
        run: |
          coverage=$(go test ./... -coverprofile=coverage.out | grep total | awk '{print $3}')
          if (( $(echo "$coverage < 80" | bc -l) )); then
            echo "Coverage is below 80%"
            exit 1
          fi
```
