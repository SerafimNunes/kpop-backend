2. **Difícil manutenção** - Não sei onde mexer

5. **Código duplicado** - Mesma lógica em vários lugares

## 🎯 SOLUÇÃO: ARQUITETURA DE DOMÍNIOS

### Princípios:

1. **Separação por domínio de negócio** (não por tipo técnico)
2. **Cada domínio é independente**
3. **Comunicação via interfaces**
4. **Testes isolados por domínio**
5. **Zero dependências circulares**

### 3. SERVICE (Lógica de Negócio)
**Responsabilidade:** Orquestrar regras de negócio

```go
// internal/domain/pgrs/service/pgrs_service.go
package service

import (
    "context"
    "fmt"
    "time"
    "auren-platform/internal/domain/pgrs/models"
    "auren-platform/internal/domain/pgrs/repository"
    "auren-platform/internal/domain/pgrs/dto"
)

type PGRSService struct {
    repo      repository.PGRSRepository
    generator *GeneratorService
    docx      *DOCXService
}

func NewPGRSService(
    repo repository.PGRSRepository,
    generator *GeneratorService,
    docx *DOCXService,
) *PGRSService {
    return &PGRSService{
        repo:      repo,
        generator: generator,
        docx:      docx,
    }
}

// Criar PGRS em estado rascunho
func (s *PGRSService) CriarRascunho(ctx context.Context, req *dto.CriarPGRSRequest) (*models.PGRS, error) {
    numero := s.gerarNumeroSequencial()
    
    pgrs := &models.PGRS{
        UnidadeID:       req.UnidadeID,
        Numero:          numero,
        Status:          models.StatusRascunho,
        DadosFormulario: req.DadosFormulario,
    }
    
    if err := s.repo.Create(ctx, pgrs); err != nil {
        return nil, fmt.Errorf("erro ao criar PGRS: %w", err)
    }
    
    return pgrs, nil
}

// Gerar PGRS com IA (assíncrono)
func (s *PGRSService) GerarComIA(ctx context.Context, pgrsID uint) error {
    // Buscar PGRS
    pgrs, err := s.repo.FindByID(ctx, pgrsID)
    if err != nil {
        return err
    }
    
    // Validar estado
    if pgrs.Status != models.StatusRascunho {
        return fmt.Errorf("PGRS deve estar em rascunho")
    }
    
    // Atualizar status
    pgrs.Status = models.StatusEmGeracao
    if err := s.repo.Update(ctx, pgrs); err != nil {
        return err
    }
    
    // Chamar gerador (delega para outro service)
    go s.generator.ProcessarGeracao(ctx, pgrs)
    
    return nil
}

// Aprovar PGRS
func (s *PGRSService) Aprovar(ctx context.Context, pgrsID uint, revisorID uint) error {
    pgrs, err := s.repo.FindByID(ctx, pgrsID)
    if err != nil {
        return err
    }
    
    if !pgrs.PodeSerAprovado() {
        return fmt.Errorf("PGRS não pode ser aprovado no status atual")
    }
    
    pgrs.Status = models.StatusAprovado
    pgrs.RevisorID = &revisorID
    now := time.Now()
    pgrs.DataAprovacao = &now
    
    return s.repo.Update(ctx, pgrs)
}

func (s *PGRSService) gerarNumeroSequencial() string {
    // Lógica para gerar PGRS-YYYY-NNN
    return fmt.Sprintf("PGRS-%d-%03d", time.Now().Year(), 1) // Simplificado
}
```

```go
// internal/domain/pgrs/handlers/http_handler.go
package handlers

import (
    "net/http"
    "strconv"
    "github.com/gin-gonic/gin"
    "auren-platform/internal/domain/pgrs/service"
    "auren-platform/internal/domain/pgrs/dto"
)

type HTTPHandler struct {
    service *service.PGRSService
}

func NewHTTPHandler(service *service.PGRSService) *HTTPHandler {
    return &HTTPHandler{service: service}
}

// POST /api/pgrs
func (h *HTTPHandler) Create(c *gin.Context) {
    var req dto.CriarPGRSRequest
    if err := c.ShouldBindJSON(&req); err != nil {
        c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
        return
    }
    
    pgrs, err := h.service.CriarRascunho(c.Request.Context(), &req)
    if err != nil {
        c.JSON(http.StatusInternalServerError, gin.H{"error": err.Error()})
        return
    }
    
    c.JSON(http.StatusCreated, pgrs)
}

// GET /api/pgrs/:id
func (h *HTTPHandler) GetByID(c *gin.Context) {
    id, _ := strconv.ParseUint(c.Param("id"), 10, 32)
    
    pgrs, err := h.service.BuscarPorID(c.Request.Context(), uint(id))
    if err != nil {
        c.JSON(http.StatusNotFound, gin.H{"error": "PGRS não encontrado"})
        return
    }
    
    c.JSON(http.StatusOK, pgrs)
}

// POST /api/pgrs/:id/gerar
func (h *HTTPHandler) GerarComIA(c *gin.Context) {
    id, _ := strconv.ParseUint(c.Param("id"), 10, 32)
    
    err := h.service.GerarComIA(c.Request.Context(), uint(id))
    if err != nil {
        c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
        return
    }
    
    c.JSON(http.StatusOK, gin.H{"message": "Geração iniciada"})
}

// RegisterRoutes registra todas as rotas HTTP do domínio
func (h *HTTPHandler) RegisterRoutes(router *gin.RouterGroup) {
    pgrs := router.Group("/pgrs")
    {
        pgrs.POST("", h.Create)
        pgrs.GET("/:id", h.GetByID)
        pgrs.POST("/:id/gerar", h.GerarComIA)
        pgrs.POST("/:id/aprovar", h.Aprovar)
        // ... outras rotas
    }
}
```

```go
// internal/domain/pgrs/handlers/ws_handler.go
package handlers

import (
    "context"
    "encoding/json"
    "log"
    "auren-platform/internal/domain/pgrs/service"
    "auren-platform/internal/shared/websocket"
)

type WSHandler struct {
    service *service.PGRSService
}

func NewWSHandler(service *service.PGRSService) *WSHandler {
    return &WSHandler{service: service}
}

// HandleMessage processa mensagens WebSocket específicas do domínio PGRS
func (h *WSHandler) HandleMessage(client *websocket.Client, message []byte) error {
    var msg websocket.Message
    if err := json.Unmarshal(message, &msg); err != nil {
        return err
    }
    
    switch msg.Action {
    case "pgrs.gerar":
        return h.handleGerar(client, msg.Payload)
    case "pgrs.aprovar":
        return h.handleAprovar(client, msg.Payload)
    default:
        log.Printf("[PGRS WS] Ação desconhecida: %s", msg.Action)
    }
    
    return nil
}

func (h *WSHandler) handleGerar(client *websocket.Client, payload json.RawMessage) error {
    var req struct {
        PGRSID uint `json:"pgrs_id"`
    }
    
    if err := json.Unmarshal(payload, &req); err != nil {
        return err
    }
    
    ctx := context.Background()
    err := h.service.GerarComIA(ctx, req.PGRSID)
    
    if err != nil {
        client.Send(websocket.Message{
            Action: "pgrs.gerar.erro",
            Payload: map[string]string{"error": err.Error()},
        })
        return err
    }
    
    client.Send(websocket.Message{
        Action: "pgrs.gerar.iniciado",
        Payload: map[string]uint{"pgrs_id": req.PGRSID},
    })
    
    return nil
}
```

## 🔄 WEBSOCKET REFATORADO

### Hub Genérico (Shared)

```go
// internal/shared/websocket/hub.go
package websocket

type Hub struct {
    clients    map[*Client]bool
    register   chan *Client
    unregister chan *Client
    broadcast  chan Message
    handlers   map[string]MessageHandler  // ← Handlers por domínio
}

type MessageHandler interface {
    HandleMessage(client *Client, message []byte) error
}

func NewHub() *Hub {
    return &Hub{
        clients:    make(map[*Client]bool),
        register:   make(chan *Client),
        unregister: make(chan *Client),
        broadcast:  make(chan Message),
        handlers:   make(map[string]MessageHandler),
    }
}

// Registrar handler de um domínio
func (h *Hub) RegisterHandler(prefix string, handler MessageHandler) {
    h.handlers[prefix] = handler
}

func (h *Hub) Run() {
    for {
        select {
        case client := <-h.register:
            h.clients[client] = true
        case client := <-h.unregister:
            delete(h.clients, client)
        case message := <-h.broadcast:
            h.broadcastMessage(message)
        }
    }
}

func (h *Hub) RouteMessage(client *Client, message []byte) error {
    var msg Message
    if err := json.Unmarshal(message, &msg); err != nil {
        return err
    }
    
    // Extrair prefixo da action (ex: "pgrs.gerar" → "pgrs")
    prefix := strings.Split(msg.Action, ".")[0]
    
    // Rotear para handler do domínio
    if handler, ok := h.handlers[prefix]; ok {
        return handler.HandleMessage(client, message)
    }
    
    return fmt.Errorf("handler não encontrado para: %s", prefix)
}
```

```go
// cmd/server/main.go
package main

import (
    "log"
    "auren-platform/config"
    "auren-platform/internal/shared/database"
    "auren-platform/internal/shared/websocket"
    "auren-platform/internal/infrastructure/gemini"
    
    // Domínios

    pgrsRepo "auren-platform/internal/domain/pgrs/repository"

    adminService "auren-platform/internal/domain/admin/service"

)

func main() {
    // 1. Carregar configurações
    cfg := config.Load()
    
    // 2. Conectar banco
    db := database.Connect(cfg.DatabaseURL)
    database.RunMigrations(db)
    
    // 3. Inicializar infraestrutura
    geminiClient := gemini.NewClient(cfg.GeminiAPIKey)
    
    // 4. Inicializar Hub WebSocket
    hub := websocket.NewHub()

    // 5. Inicializar domínios
    initPGRSDomain(db, geminiClient, hub)
    initAdminDomain(db, hub)
    // ... outros domínios
    
    // 6. Inicializar servidor HTTP
    router := gin.Default()
    router.GET("/ws", func(c *gin.Context) {
        websocket.ServeWS(hub, c.Writer, c.Request)
    })
    
    // 7. Iniciar servidor
    log.Fatal(router.Run(":" + cfg.Port))
}

func initPGRSDomain(db *gorm.DB, gemini *gemini.Client, hub *websocket.Hub) {
    // Repository
    pgrsRepository := pgrsRepo.NewPGRSRepository(db)
    
    // Services
    generatorSvc := pgrsService.NewGeneratorService(gemini)
    docxSvc := pgrsService.NewDOCXService()
    pgrsSvc := pgrsService.NewPGRSService(pgrsRepository, generatorSvc, docxSvc)
    
    // Handlers
    httpHandler := pgrsHandlers.NewHTTPHandler(pgrsSvc)
    wsHandler := pgrsHandlers.NewWSHandler(pgrsSvc)
    
    // Registrar rotas HTTP
    router.Group("/api").Use(/* middlewares */)
    httpHandler.RegisterRoutes(router.Group("/api"))
    
    // Registrar handler WebSocket
    hub.RegisterHandler("pgrs", wsHandler)
}
```
