```html
function dashboardModule() {
  return {
    kpis: {
      ativos: 0,

      taxaAprovacao: 0
    },
    receitaMensal: 0,
    variacao: 0,

    init() {
      this.carregarDados();
      
      // Atualizar a cada 30s
      setInterval(() => this.carregarDados(), 30000);
    },
    
    async carregarDados() {
      try {
        const response = await fetch('/api/dashboard/kpis');
        const data = await response.json();
        
        this.kpis = data.kpis;
        this.receitaMensal = data.receita_mensal;
```

**Backend - Criar handler `/api/dashboard/kpis`:**

```go
// internal/handlers/dashboard.go
package handlers

import (
    "auren-platform/db"
    "github.com/gin-gonic/gin"
    "time"
)

func HandleDashboardKPIs(c *gin.Context) {
    var ativos, pendentes, concluidos int64
    
    // PGRS Ativos
    db.DB.Model(&db.PGRS{}).
        Where("status NOT IN ?", []string{"ARQUIVADO", "REJEITADO"}).
        Count(&ativos)
    
    // Pendentes de Revisão
    db.DB.Model(&db.PGRS{}).
        Where("status = ?", "AGUARDANDO_REVISAO").
        Count(&pendentes)
    
    // Concluídos no mês
    inicioMes := time.Now().AddDate(0, 0, -time.Now().Day()+1)
    db.DB.Model(&db.PGRS{}).
        Where("status = ? AND updated_at >= ?", "APROVADO", inicioMes).
        Count(&concluidos)
    
    // Taxa de aprovação (últimos 30 dias)
    var total, aprovados int64
    trintaDiasAtras := time.Now().AddDate(0, 0, -30)
    
    db.DB.Model(&db.PGRS{}).
        Where("created_at >= ?", trintaDiasAtras).
        Count(&total)
    
    db.DB.Model(&db.PGRS{}).
        Where("status = ? AND created_at >= ?", "APROVADO", trintaDiasAtras).
        Count(&aprovados)
    
    taxaAprovacao := 0.0
    if total > 0 {
        taxaAprovacao = float64(aprovados) / float64(total) * 100
    }
    
    // Receita mensal (placeholder - ajustar conforme necessário)
    var receitaMensal float64
    db.DB.Model(&db.LancamentoFinanceiro{}).
        Select("COALESCE(SUM(amount), 0)").
        Where("type = 'entrada' AND EXTRACT(MONTH FROM date) = ?", time.Now().Month()).
        Scan(&receitaMensal)
    
    c.JSON(200, gin.H{
        "kpis": gin.H{
            "ativos": ativos,
            "pendentes": pendentes,
            "concluidos": concluidos,
            "taxaAprovacao": int(taxaAprovacao),
        },
        "receita_mensal": receitaMensal,
        "variacao": 0, // TODO: Calcular variação
        "alertas": []interface{}{}, // TODO: Implementar alertas
        "atividades": []interface{}{}, // TODO: Implementar atividades
    })
}
```

```javascript
      treinamento: { frequencia: '', conteudo: '', registro: '' }
    },
    
    // Controles
    saving: false,
    errors: {},
    
    init() {
      this.carregarDraft();
      this.iniciarAutoSave();
    },
    
    // Navegação
    proximoPasso() {
      if (this.validarPasso(this.currentStep)) {
        this.currentStep++;
        this.salvarDraft();
        window.scrollTo(0, 0);
      }
    },
    
    passoAnterior() {
      if (this.currentStep > 1) {
        this.currentStep--;
        window.scrollTo(0, 0);
      }
    },
    
    irParaPasso(step) {
      if (step <= this.currentStep || this.validarPasso(this.currentStep)) {
        this.currentStep = step;
        window.scrollTo(0, 0);
      }
    },
    
    // Validações
    validarPasso(step) {
      this.errors = {};
      
      switch(step) {
        case 1:
          if (!this.empresa.razao_social) {
            this.errors.razao_social = 'Razão Social é obrigatória';
          }
          if (!this.validarCNPJ(this.empresa.cnpj)) {
            this.errors.cnpj = 'CNPJ inválido';
          }
          break;
          
        case 2:
          if (!this.unidade.nome) {
            this.errors.unidade_nome = 'Nome da unidade é obrigatório';
          }
          if (!this.unidade.responsavel_tecnico) {
            this.errors.responsavel_tecnico = 'Responsável Técnico é obrigatório';
          }
          break;
          
        case 3:
          if (this.residuos.length === 0) {
            this.errors.residuos = 'Adicione pelo menos um resíduo';
          }
          break;
      }
      
      return Object.keys(this.errors).length === 0;
    },
    
    validarCNPJ(cnpj) {
      cnpj = cnpj.replace(/[^\d]/g, '');
      if (cnpj.length !== 14) return false;
      
      // Algoritmo de validação completo
      // ... (implementar)
      
      return true;
    },
    
    // Resíduos
    adicionarResiduo() {
      this.residuos.push({
        nome: '',
        classe: '',
        quantidade_mensal_kg: null,
        destinacao: '',
        observacoes: ''
      });
    },
    
    removerResiduo(index) {
      this.residuos.splice(index, 1);
    },
    
    // Auto-save
    iniciarAutoSave() {
      setInterval(() => {
        this.salvarDraft();
      }, 30000); // A cada 30s
    },
    
    salvarDraft() {
      const draft = {
        empresa: this.empresa,
        unidade: this.unidade,
        residuos: this.residuos,
        operacional: this.operacional,
        currentStep: this.currentStep
      };
      
      localStorage.setItem('pgrs_draft', JSON.stringify(draft));
    },
    
    carregarDraft() {
      const draft = localStorage.getItem('pgrs_draft');
      if (draft) {
        const data = JSON.parse(draft);
        this.empresa = data.empresa || this.empresa;
        this.unidade = data.unidade || this.unidade;
        this.residuos = data.residuos || [];
        this.operacional = data.operacional || this.operacional;
        this.currentStep = data.currentStep || 1;
      }
    },
    
    limparDraft() {
      localStorage.removeItem('pgrs_draft');
    },
    
    // Submissão
    async gerarPGRS() {
      this.saving = true;
      
      const dados = {
        empresa: this.empresa,
        unidade: this.unidade,
        inventario: this.residuos,
        operacional: this.operacional
      };
      
      try {
        const response = await fetch('/api/pgrs/gerar', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(dados)
        });
        
        if (!response.ok) throw new Error('Erro ao gerar PGRS');
        
        const { pgrs_id } = await response.json();
        
        this.limparDraft();
        window.location.href = `#engenharia/pgrs/${pgrs_id}`;
        
      } catch (error) {
        alert('Erro ao gerar PGRS: ' + error.message);
      } finally {
        this.saving = false;
      }
    },
    
    // Utilidades
    formatarCNPJ(valor) {
      valor = valor.replace(/\D/g, '');
      return valor.replace(/^(\d{2})(\d{3})(\d{3})(\d{4})(\d{2})$/, '$1.$2.$3/$4-$5');
    },
    
    progresso() {
      return (this.currentStep / this.totalSteps) * 100;
    }
  };
}
```
